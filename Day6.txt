Constructors
-----------------------------------------------------------------------------
class Food{
	private String name;
	private Integer category;
	private String ingredients;
	
	//default constructor
	Food(){}
	
	//parameterized constructor
	Food(String name, Integer category, String ingredients){
		this();
		this.name = name;
		this.category = category;
		this.ingredients = ingredients;
	}
	
	//copy constructor
	Food(Food newFood){
		this.name = newFood.name;
		this.category = newFood.category;
		this.ingredients = newFood.ingredients;
	}
}
class Main{
	public static void main(String[] a){
		Food food = new Food();
	}
}
